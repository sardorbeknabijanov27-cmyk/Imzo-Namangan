// doGet — barcha so'rovlar shu funksiyadan kiradi
// action=log bo'lsa tashrif yozadi, aks holda ma'lumot qaytaradi
function doGet(e) {
  const action = e && e.parameter && e.parameter.action;
  if (action === 'log') return logVisit(e);
  return getData();
}

// logVisit — kimdir dashboard'ni ochganda "Tashrif" varag'iga yozadi
// Yozadi: vaqt, qurilma turi, model, OS, brauzer, ekran, til, vaqt zonasi
function logVisit(e) {
  const ss = SpreadsheetApp.openById('1Kiw5YJZWmlkHMbjvlgcut9XsKP3NuiKKUuC_toxtEaE');
  let logSheet = ss.getSheetByName('Tashrif');
  if (!logSheet) {
    logSheet = ss.insertSheet('Tashrif');
    logSheet.appendRow(['Vaqt','Qurilma','Model','OS','Brauzer','Ekran','Til','Vaqt zonasi']);
    logSheet.getRange(1,1,1,8).setFontWeight('bold').setBackground('#1a1a2e').setFontColor('#ffffff');
    logSheet.setFrozenRows(1);
    logSheet.setColumnWidth(1,160); logSheet.setColumnWidth(2,110);
    logSheet.setColumnWidth(3,170); logSheet.setColumnWidth(4,120);
    logSheet.setColumnWidth(5,110); logSheet.setColumnWidth(6,140);
  }
  const p = e.parameter || {};
  const timeStr = Utilities.formatDate(new Date(), 'Asia/Tashkent', 'dd.MM.yyyy HH:mm:ss');
  logSheet.appendRow([timeStr, p.device||'—', p.model||'—', p.os||'—',
                      p.browser||'—', p.screen||'—', p.lang||'—', p.tz||'—']);
  return ContentService.createTextOutput(JSON.stringify({ok:true}))
    .setMimeType(ContentService.MimeType.JSON);
}

// getData — "Касса" varag'idan xarajatlarni o'qib JSON qaytaradi
// Sheets tuzilmasi:
//   5-qator: sarlavhalar
//   6-qatordan: ma'lumotlar
//   Ustun 5 (F): сана (sana)
//   Ustun 6 (G): изох (tavsif)
//   Ustun 7 (H): код (kategoriya raqami)
//   Ustun 8 (I): сумма
//   Ustun 12 (M): kategoriya raqami (jadval)
//   Ustun 13 (N): kategoriya nomi (jadval)
function getData() {
  const SHEET_ID = '1Kiw5YJZWmlkHMbjvlgcut9XsKP3NuiKKUuC_toxtEaE';
  const GID = 897386677;
  try {
    const ss = SpreadsheetApp.openById(SHEET_ID);
    const sheet = ss.getSheets().find(s => s.getSheetId() === GID);
    const data = sheet.getDataRange().getValues();
    const rows = data.slice(5);
    // Kategoriya kodi → nomi lug'ati
    const catMap = {};
    rows.forEach(row => {
      const num = row[12], name = String(row[13]||'').trim();
      if (num && name) catMap[String(num).trim()] = name;
    });
    const operations = [], catSet = new Set();
    rows.forEach(row => {
      const amount = parseFloat(String(row[8]||'').replace(/\s/g,'').replace(',','.')) || 0;
      if (!amount) return;
      const day = formatDay(row[5]); if (!day) return;
      const cat = catMap[String(row[7]||'').trim()] || 'Boshqa';
      const desc = String(row[6]||'').trim();
      operations.push({day, cat, desc, amount}); catSet.add(cat);
    });
    operations.sort((a,b) => a.day.localeCompare(b.day));
    return out({operations, categories:[...catSet].sort()});
  } catch(err) { return out({error:err.toString(), operations:[], categories:[]}); }
}

// formatDay — turli formatdagi sanani YYYY-MM-DD ga o'tkazadi
function formatDay(val) {
  if (!val) return null;
  if (val instanceof Date) {
    return val.getFullYear()+'-'+String(val.getMonth()+1).padStart(2,'0')+
           '-'+String(val.getDate()).padStart(2,'0');
  }
  const s = String(val).trim();
  const m = s.match(/^(\d{1,2})\.(\d{1,2})\.(\d{4})$/);
  if (m) return `${m[3]}-${m[2].padStart(2,'0')}-${m[1].padStart(2,'0')}`;
  if (/^\d{4}-\d{2}-\d{2}$/.test(s)) return s;
  return null;
}

function out(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
